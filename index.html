    <script src="https://unpkg.com/@jspsych/plugin-html-keyboard-response@1.1.3"></script>

    <script src="https://unpkg.com/@jspsych/plugin-image-keyboard-response@1.1.3"></script>

    <script src="https://unpkg.com/@jspsych/plugin-audio-keyboard-response@1.1.3"></script>

    <script src="https://unpkg.com/@jspsych/plugin-preload@1.1.3"></script>

    <link href="https://unpkg.com/jspsych@7.3.4/css/jspsych.css" rel="stylesheet" type="text/css" />

    <style>

        .stim-container { display: flex; justify-content: space-around; align-items: center; width: 800px; }

        .stim-image { width: 300px; height: auto; }

        .fixation { font-size: 60px; font-weight: bold; }

    </style>

<script>

    const jsPsych = initJsPsych({

        on_finish: function() {

            jsPsych.data.displayData(); // For testing: shows data at the end

        }

    });


    // 1. Define Stimuli (This would typically be a larger array or imported JSON)

    const timeline_variables = [

        { 

            prime: 'assets/primes/prime_a.jpg', 

            audio: 'assets/audio/mandarin_word1.mp3', 

            left_img: 'assets/stims/target1.jpg', 

            right_img: 'assets/stims/distractor1.jpg',

            correct_response: 'f' 

        }

        // Add more trials here

    ];


    // 2. Preload Assets

    const preload = {

        type: jsPsychPreload,

        auto_preload: true

    };


    // 3. Fixation Cross

    const fixation = {

        type: jsPsychHtmlKeyboardResponse,

        stimulus: '<div class="fixation">+</div>',

        choices: "NO_KEYS",

        trial_duration: 500,

    };


    // 4. Prime Image

    const prime_trial = {

        type: jsPsychImageKeyboardResponse,

        stimulus: jsPsych.timelineVariable('prime'),

        choices: "NO_KEYS",

        trial_duration: 500,

        // Optional: logic to skip this if a 'no prime' condition is met

    };


    // 5. Audio + Two-Image Forced Choice

    const test_trial = {

        type: jsPsychAudioKeyboardResponse,

        stimulus: jsPsych.timelineVariable('audio'),

        choices: ['f', 'j'],

        prompt: function() {

            return `

                <div class="stim-container">

                    <img src="${jsPsych.timelineVariable('left_img')}" class="stim-image">

                    <img src="${jsPsych.timelineVariable('right_img')}" class="stim-image">

                </div>

                <p>Press 'F' for Left or 'J' for Right</p>`;

        },

        trial_duration: 4000,

        response_ends_trial: true,

        data: {

            task: 'response',

            correct_answer: jsPsych.timelineVariable('correct_response')

        }

    };


    // 6. Assemble Timeline

    const main_procedure = {

        timeline: [fixation, prime_trial, test_trial],

        timeline_variables: timeline_variables,

        randomize_order: true

    };


    jsPsych.run([preload, main_procedure]);


</script>
